```tsx
import { Users, Wifi, DollarSign, TrendingUp } from 'lucide-react'
import { useMetrics } from '@/hooks/useMetrics'
import { MetricCard } from './MetricCard'
import { MetricCardSkeleton } from './MetricCardSkeleton'
import type { MetricsData } from '@/types/api'

// ── Formateo de pesos colombianos ──
// Formato: $ 9.551.300 (punto como separador de miles, sin decimales para enteros)
function formatCOP(amount: number): string {
  // Usar Intl.NumberFormat con locale 'es-CO' produce "$ 9.551.300"
  return new Intl.NumberFormat('es-CO', {
    style: 'currency',
    currency: 'COP',
    minimumFractionDigits: 0,
    maximumFractionDigits: 0,
  }).format(amount)
}

function formatNumber(n: number): string {
  return new Intl.NumberFormat('es-CO').format(n)
}

interface MetricsGridProps {
  // Permite inyectar data en el futuro para testing — por defecto usa el hook
  _data?: MetricsData
}

export function MetricsGrid({ _data }: MetricsGridProps = {}) {
  const { data, isLoading, isError, error } = useMetrics()

  // Estado de carga — mostrar 4 skeletons
  if (isLoading) {
    return (
      <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
        <MetricCardSkeleton />
        <MetricCardSkeleton />
        <MetricCardSkeleton />
        <MetricCardSkeleton />
      </div>
    )
  }

  // Estado de error
  if (isError) {
    return (
      <div className="rounded-xl border border-error/30 bg-error/10 px-6 py-4">
        <p className="text-sm font-medium text-error">
          Error al cargar métricas
        </p>
        <p className="mt-1 text-xs text-text-muted">
          {error?.message ?? 'No se pudieron obtener los datos de Wisphub'}
        </p>
      </div>
    )
  }

  const metrics = _data ?? data

  if (!metrics) return null

  return (
    <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-4">
      {/* MET-01: Clientes activos */}
      <MetricCard
        title="Clientes activos"
        value={formatNumber(metrics.activeClients)}
        subtitle="Estado Activo en Wisphub"
        icon={Users}
        iconClassName="text-text-secondary"
      />

      {/* MET-02: Clientes conectados — SIEMPRE N/D */}
      <MetricCard
        title="Conectados ahora"
        value="N/D"
        subtitle="No disponible en Wisphub API"
        icon={Wifi}
        iconClassName="text-text-muted"
      />

      {/* MET-03: Deuda pendiente */}
      <MetricCard
        title="Deuda pendiente"
        value={formatCOP(metrics.pendingDebt)}
        subtitle="Saldo positivo total"
        icon={DollarSign}
        iconClassName="text-brand"
      />

      {/* MET-04: Ingresos del mes */}
      <MetricCard
        title="Ingresos del mes"
        value={formatCOP(metrics.monthlyRevenue)}
        subtitle="Facturas pagadas este mes"
        icon={TrendingUp}
        iconClassName="text-brand"
      />
    </div>
  )
}
```
