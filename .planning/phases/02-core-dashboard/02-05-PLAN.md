```tsx
import { useState, useCallback } from 'react'
import { RefreshCw } from 'lucide-react'
import { useQueryClient, useIsFetching } from '@tanstack/react-query'
import { cn } from '@/lib/utils'
import { Button } from '@/components/ui/button'
import { METRICS_QUERY_KEY } from '@/hooks/useMetrics'
import { INVOICES_QUERY_KEY } from '@/hooks/useInvoices'

const COOLDOWN_MS = 30_000  // 30 segundos — rate limit protection

export function RefreshButton() {
  const queryClient = useQueryClient()
  const isFetching = useIsFetching()
  const [cooldownActive, setCooldownActive] = useState(false)

  const isDisabled = isFetching > 0 || cooldownActive
  const isLoading = isFetching > 0

  const handleRefresh = useCallback(async () => {
    if (isDisabled) return

    // Activar cooldown inmediatamente — el usuario no puede hacer spam
    setCooldownActive(true)

    // Invalidar ambas queries para refetch simultáneo
    await Promise.all([
      queryClient.invalidateQueries({ queryKey: METRICS_QUERY_KEY }),
      queryClient.invalidateQueries({ queryKey: INVOICES_QUERY_KEY }),
    ])

    // Desactivar cooldown después de 30 segundos
    setTimeout(() => {
      setCooldownActive(false)
    }, COOLDOWN_MS)
  }, [isDisabled, queryClient])

  return (
    <Button
      variant="outline"
      size="sm"
      onClick={handleRefresh}
      disabled={isDisabled}
      className={cn(
        'border-border bg-surface text-text-secondary',
        'hover:border-brand hover:text-brand',
        'disabled:opacity-50 disabled:cursor-not-allowed',
        'transition-colors',
      )}
      aria-label="Actualizar datos del dashboard"
    >
      <RefreshCw
        className={cn(
          'mr-2 h-4 w-4',
          isLoading && 'animate-spin',
        )}
        aria-hidden="true"
      />
      {isLoading ? 'Actualizando...' : cooldownActive ? 'Espera 30s' : 'Actualizar'}
    </Button>
  )
}
```

```tsx
import { useMetrics } from '@/hooks/useMetrics'

function formatTimestamp(ms: number): string | null {
  if (ms === 0) return null

  const fecha = new Date(ms)

  // Formato: "Hoy a las 14:32" o "07/04/2026 14:32"
  const hoy = new Date()
  const esHoy =
    fecha.getDate() === hoy.getDate() &&
    fecha.getMonth() === hoy.getMonth() &&
    fecha.getFullYear() === hoy.getFullYear()

  const hora = fecha.toLocaleTimeString('es-CO', {
    hour: '2-digit',
    minute: '2-digit',
    hour12: false,
  })

  if (esHoy) {
    return `Hoy a las ${hora}`
  }

  const fechaStr = fecha.toLocaleDateString('es-CO', {
    day: '2-digit',
    month: '2-digit',
    year: 'numeric',
  })

  return `${fechaStr} ${hora}`
}

export function LastUpdated() {
  const { dataUpdatedAt } = useMetrics()
  const label = formatTimestamp(dataUpdatedAt)

  if (!label) return null

  return (
    <p className="text-xs text-text-muted" aria-live="polite">
      Última actualización: {label}
    </p>
  )
}
```

## Trust Boundaries

|----------|-------------|
| User click → invalidateQueries | Input del usuario desencadena múltiples requests a BFF |
| setTimeout cooldown | Estado de UI controlado por timer — no persiste entre recargas |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-13 | Denial of Service | RefreshButton spam clicks | mitigate | Cooldown de 30s después del primer click. isDisabled = isFetching > 0 || cooldownActive previene clicks durante carga. El BFF también tiene cache de 30s como segunda línea de defensa. |
| T-02-14 | Tampering | cooldown timer bypass | accept | El cooldown es solo UX — no es una medida de seguridad. El BFF cache de 30s es la protección real de rate limit. Un usuario técnico puede bypassear el cooldown via DevTools, pero el BFF los ignorará por el cache. |
| T-02-15 | Information Disclosure | LastUpdated timestamp | accept | El timestamp muestra cuándo se actualizó por última vez — no es información sensible. Solo el admin ve este dashboard. |
</threat_model>
